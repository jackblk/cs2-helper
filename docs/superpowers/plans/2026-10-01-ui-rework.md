# Main-Window UI Rework Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the Save/Discard main page with an autosaving, undoable, shift-free UI (status bar, setup dialog, sectioned settings) and move overlay positioning into an in-overlay edit mode.

**Architecture:** A framework-free settings store (`src/lib/settings-store.ts`: serialized save queue + undo/redo history) is wrapped by a React `SettingsProvider` in `Layout`, so every control commits through one `commit(label, update)` call. `MainPage` becomes a composition of small section components sharing one row grid. Overlay editing turns the overlay window into a full-monitor surface positioned in JS; Rust only resizes the window and reports the result, which the main window saves like any other setting.

**Tech Stack:** Tauri v2 (Rust), React 19, TypeScript, Vite 8, Vitest 4, Tailwind v4, shadcn/ui (radix-nova), sonner 2, lucide-react, pnpm, Biome.

**Spec:** `docs/superpowers/specs/2026-10-01-ui-rework-design.md`

## Review Focus

* A save fails in the middle of a burst of edits (hotkey taken by another app): the UI must snap back to the last saved state, drop the queued edits built on the failed one, show one error toast, and record nothing in history. Pinned in Task 1 (`settings-store.test.ts`, "reverts and drops queued commits").
* An undo or redo whose save fails: state and both stacks must stay exactly as they were. Pinned in Task 1 ("leaves the stack and state alone when an undo save fails").
* A number field receives blank text, `0`, a negative, or `0.3` for an integer ms field: nothing is saved and the field reverts. Pinned in Task 2 (`parse.test.ts`).
* The overlay is dragged to a screen edge, or shrunk to 0.5x where the toolbar is wider than the overlay: the toolbar stays fully on screen. Pinned in Task 2 (`overlay-edit.test.ts`).
* The tray reloads the config while saves are queued or in flight: the form shows the reloaded settings, history is cleared, and no stale save lands afterwards. Pinned in Task 1 ("replace clears history and discards queued and in-flight saves").

```ts
export type HistoryEntry<T> = { label: string; before: T; after: T };

export type History<T> = {
	past: HistoryEntry<T>[];
	future: HistoryEntry<T>[];
};

export const HISTORY_CAP = 50;

export function emptyHistory<T>(): History<T> {
	return { past: [], future: [] };
}

export function record<T>(h: History<T>, entry: HistoryEntry<T>): History<T> {
	return { past: [...h.past, entry].slice(-HISTORY_CAP), future: [] };
}

export function undone<T>(h: History<T>): History<T> {
	if (h.past.length === 0) return h;
	const entry = h.past[h.past.length - 1];
	return { past: h.past.slice(0, -1), future: [entry, ...h.future] };
}

export function redone<T>(h: History<T>): History<T> {
	if (h.future.length === 0) return h;
	const [entry, ...rest] = h.future;
	return { past: [...h.past, entry], future: rest };
}
```

```ts
import {
	emptyHistory,
	type History,
	record,
	redone,
	undone,
} from "@/lib/history";

export type StoreNotice =
	| { kind: "saved" | "undone" | "redone" | "applied"; label: string }
	| { kind: "failed"; title: string; error: unknown };

export type StoreSnapshot<T> = { current: T | null; history: History<T> };

const same = (a: unknown, b: unknown) =>
	JSON.stringify(a) === JSON.stringify(b);

// Autosave with undo/redo. `current` is shown optimistically; `saved` is the
// last state the backend accepted. Jobs run one at a time in commit order, so
// a slow save can never land after a newer one.
export function createSettingsStore<T>({
	save,
	notify,
}: {
	save: (value: T) => Promise<void>;
	notify: (notice: StoreNotice) => void;
}) {
	let snapshot: StoreSnapshot<T> = {
		current: null,
		history: emptyHistory<T>(),
	};
	let saved: T | null = null;
	// Bumped when queued jobs no longer apply: a failed save they were built
	// on, or settings replaced from outside.
	let generation = 0;
	let tail: Promise<void> = Promise.resolve();
	const listeners = new Set<() => void>();

	const set = (patch: Partial<StoreSnapshot<T>>) => {
		snapshot = { ...snapshot, ...patch };
		for (const listener of listeners) listener();
	};

	const enqueue = (job: (stale: () => boolean) => Promise<void>) => {
		const gen = generation;
		const stale = () => gen !== generation;
		tail = tail.then(() => (stale() ? undefined : job(stale)));
		return tail;
	};

	const fail = (title: string, error: unknown) => {
		generation += 1;
		set({ current: saved });
		notify({ kind: "failed", title, error });
	};

	return {
		getSnapshot: () => snapshot,

		subscribe(listener: () => void) {
			listeners.add(listener);
			return () => {
				listeners.delete(listener);
			};
		},

		replace(value: T) {
			generation += 1;
			saved = value;
			set({ current: value, history: emptyHistory<T>() });
		},

		commit(label: string, update: (value: T) => T): Promise<void> {
			const before = snapshot.current;
			if (before == null) return tail;
			const after = update(before);
			if (same(before, after)) return tail;
			set({ current: after });
			return enqueue(async (stale) => {
				try {
					await save(after);
				} catch (e) {
					if (!stale()) fail("Couldn't save", e);
					return;
				}
				if (stale()) return;
				saved = after;
				set({ history: record(snapshot.history, { label, before, after }) });
				notify({ kind: "saved", label });
			});
		},

		run(label: string, failTitle: string, op: () => Promise<T>): Promise<void> {
			return enqueue(async (stale) => {
				const before = saved;
				let after: T;
				try {
					after = await op();
				} catch (e) {
					if (!stale()) fail(failTitle, e);
					return;
				}
				if (stale()) return;
				saved = after;
				set({
					current: after,
					history:
						before == null
							? snapshot.history
							: record(snapshot.history, { label, before, after }),
				});
				notify({ kind: "applied", label });
			});
		},

		undo(): Promise<void> {
			return enqueue(async (stale) => {
				const { past } = snapshot.history;
				if (past.length === 0) return;
				const entry = past[past.length - 1];
				try {
					await save(entry.before);
				} catch (e) {
					if (!stale())
						notify({ kind: "failed", title: "Couldn't undo", error: e });
					return;
				}
				if (stale()) return;
				saved = entry.before;
				set({ current: entry.before, history: undone(snapshot.history) });
				notify({ kind: "undone", label: entry.label });
			});
		},

		redo(): Promise<void> {
			return enqueue(async (stale) => {
				const { future } = snapshot.history;
				if (future.length === 0) return;
				const entry = future[0];
				try {
					await save(entry.after);
				} catch (e) {
					if (!stale())
						notify({ kind: "failed", title: "Couldn't redo", error: e });
					return;
				}
				if (stale()) return;
				saved = entry.after;
				set({ current: entry.after, history: redone(snapshot.history) });
				notify({ kind: "redone", label: entry.label });
			});
		},

		idle: () => tail,
	};
}

export type SettingsStore<T> = ReturnType<typeof createSettingsStore<T>>;
```

```ts
import type { Rect } from "@/lib/types";

export type Size = { width: number; height: number };

// Mirrors BASE_W / BASE_H in src-tauri/src/overlay.rs.
export const OVERLAY_BASE: Size = { width: 240, height: 96 };
// Mirrors `scale` in src-tauri/src/config/default.toml.
export const DEFAULT_OVERLAY_SCALE = 1;

export const SCALE_MIN = 0.5;
export const SCALE_MAX = 2.5;
export const SCALE_STEP = 0.05;

// Four size-8 icon buttons, gap-1, p-1.
export const TOOLBAR_SIZE: Size = { width: 148, height: 40 };
export const TOOLBAR_GAP = 8;

export function overlaySize(scale: number): Size {
	return {
		width: OVERLAY_BASE.width * scale,
		height: OVERLAY_BASE.height * scale,
	};
}

export function stepScale(scale: number, dir: 1 | -1): number {
	const next = Math.round((scale + dir * SCALE_STEP) * 100) / 100;
	return Math.min(SCALE_MAX, Math.max(SCALE_MIN, next));
}

// `hi < lo` (item larger than the room) pins to `lo`.
const clamp = (v: number, lo: number, hi: number) =>
	Math.min(Math.max(v, lo), Math.max(lo, hi));

export function clampToMonitor(r: Rect, m: Rect): Rect {
	return {
		...r,
		x: clamp(r.x, m.x, m.x + m.width - r.width),
		y: clamp(r.y, m.y, m.y + m.height - r.height),
	};
}

export function placeToolbar(
	overlay: Rect,
	monitor: Rect,
	toolbar: Size = TOOLBAR_SIZE,
	gap: number = TOOLBAR_GAP,
): { x: number; y: number; side: "above" | "below" } {
	const x = clamp(
		overlay.x + (overlay.width - toolbar.width) / 2,
		monitor.x,
		monitor.x + monitor.width - toolbar.width,
	);
	const above = overlay.y - gap - toolbar.height;
	if (above >= monitor.y) return { x, y: above, side: "above" };
	return { x, y: overlay.y + overlay.height + gap, side: "below" };
}
```

```rust
//! Transparent, click-through, always-on-top window for the bomb defuse timer.
//! Exists while `OverlayConfig::enabled`, or for the length of an edit.
//! Geometry is in logical pixels.

use std::sync::Mutex;

use serde::{Deserialize, Serialize};
use tauri::{
    Emitter, LogicalPosition, LogicalSize, Manager, WebviewUrl, WebviewWindow,
    WebviewWindowBuilder,
};

use crate::config::OverlayConfig;

pub const OVERLAY_LABEL: &str = "overlay";
/// Logical size at scale 1.0; the badge fills it.
const BASE_W: f64 = 240.0;
const BASE_H: f64 = 96.0;

fn logical_size(scale: f32) -> (f64, f64) {
    let s = scale as f64;
    (BASE_W * s, BASE_H * s)
}

/// Logical px between the screen edge and the default bottom-center position.
const EDGE_MARGIN: f64 = 24.0;

#[derive(Debug, Clone, Copy, Serialize)]
pub struct Rect {
    pub x: f64,
    pub y: f64,
    pub width: f64,
    pub height: f64,
}

/// `x`/`y`/`scale` are the overlay's geometry when the edit began.
#[derive(Debug, Clone, Copy, Serialize)]
pub struct EditSession {
    pub monitor: Rect,
    pub x: f64,
    pub y: f64,
    pub scale: f32,
}

#[derive(Debug, Clone, Copy, Serialize, Deserialize)]
pub struct EditResult {
    pub x: f64,
    pub y: f64,
    pub scale: f32,
}

/// While set, the window covers its monitor, so `reconcile` must neither
/// shrink nor close it. Kept here, not only sent as an event, because an
/// overlay page created for the edit loads after the event fired.
static EDIT_SESSION: Mutex<Option<EditSession>> = Mutex::new(None);

pub fn edit_session() -> Option<EditSession> {
    *EDIT_SESSION.lock().unwrap()
}

/// Saved or default top-left.
fn origin(win: &WebviewWindow, cfg: &OverlayConfig) -> tauri::Result<(f64, f64)> {
    if let (Some(x), Some(y)) = (cfg.pos_x, cfg.pos_y) {
        return Ok((x, y));
    }
    let (w, h) = logical_size(cfg.scale);
    Ok(match win.primary_monitor()? {
        Some(mon) => {
            let sf = mon.scale_factor();
            let mon_w = mon.size().width as f64 / sf;
            let mon_h = mon.size().height as f64 / sf;
            (
                ((mon_w - w) / 2.0).max(0.0),
                (mon_h - h - EDGE_MARGIN).max(0.0),
            )
        }
        None => (100.0, 100.0),
    })
}

fn position(win: &WebviewWindow, cfg: &OverlayConfig) -> tauri::Result<()> {
    let (w, h) = logical_size(cfg.scale);
    let (x, y) = origin(win, cfg)?;
    win.set_size(LogicalSize::new(w, h))?;
    win.set_position(LogicalPosition::new(x, y))?;
    Ok(())
}

fn create(app: &tauri::AppHandle, cfg: &OverlayConfig) -> tauri::Result<()> {
    if app.get_webview_window(OVERLAY_LABEL).is_some() {
        return Ok(());
    }
    let (w, h) = logical_size(cfg.scale);
    let win = WebviewWindowBuilder::new(app, OVERLAY_LABEL, WebviewUrl::App("index.html".into()))
        .title("CS2 Helper Overlay")
        .inner_size(w, h)
        .transparent(true)
        .decorations(false)
        .always_on_top(true)
        .skip_taskbar(true)
        .focusable(false)
        .shadow(false)
        .resizable(false)
        .visible(true)
        .build()?;
    win.set_ignore_cursor_events(true)?;
    position(&win, cfg)?;
    Ok(())
}

pub fn reconcile(app: &tauri::AppHandle, cfg: &OverlayConfig) {
    if edit_session().is_some() {
        // `edit_end` applies the final geometry and visibility.
        let _ = app.emit("overlay:config", cfg);
        return;
    }
    match (cfg.enabled, app.get_webview_window(OVERLAY_LABEL)) {
        (true, None) => {
            let _ = create(app, cfg);
        }
        (false, Some(win)) => {
            let _ = win.close();
        }
        (true, Some(win)) => {
            let _ = win.show();
            let _ = win.set_always_on_top(true);
            let _ = position(&win, cfg);
        }
        (false, None) => {}
    }
    let _ = app.emit("overlay:config", cfg);
}

/// Watchdog repair, run about once a second on the main thread. The OS can hide
/// the overlay or drop it behind a fullscreen game without destroying it, which
/// `reconcile` never notices. Deliberately does not reposition: that would
/// fight an edit in progress.
pub fn ensure_alive(app: &tauri::AppHandle, cfg: &OverlayConfig) {
    if !cfg.enabled {
        return;
    }
    match app.get_webview_window(OVERLAY_LABEL) {
        None => {
            let _ = create(app, cfg);
        }
        Some(win) => {
            if !win.is_visible().unwrap_or(true) {
                let _ = win.show();
            }
            // Focus-safe (SWP_NOACTIVATE). Re-claims topmost even when the
            // window is still "visible" but the game has covered it.
            let _ = win.set_always_on_top(true);
        }
    }
}

fn monitor_rect(win: &WebviewWindow) -> Result<Rect, String> {
    let mon = match win.current_monitor().map_err(|e| e.to_string())? {
        Some(m) => m,
        None => win
            .primary_monitor()
            .map_err(|e| e.to_string())?
            .ok_or("no monitor found")?,
    };
    let sf = mon.scale_factor();
    Ok(Rect {
        x: mon.position().x as f64 / sf,
        y: mon.position().y as f64 / sf,
        width: mon.size().width as f64 / sf,
        height: mon.size().height as f64 / sf,
    })
}

/// Covers the overlay's monitor so the page can drag and resize the badge
/// in JS. Creates the window when the overlay is disabled.
pub fn edit_start(app: &tauri::AppHandle, cfg: &OverlayConfig) -> Result<(), String> {
    if edit_session().is_some() {
        return Ok(());
    }
    if app.get_webview_window(OVERLAY_LABEL).is_none() {
        create(app, cfg).map_err(|e| e.to_string())?;
    }
    let win = app
        .get_webview_window(OVERLAY_LABEL)
        .ok_or("overlay window is not open")?;
    let (x, y) = origin(&win, cfg).map_err(|e| e.to_string())?;
    let monitor = monitor_rect(&win)?;
    let session = EditSession {
        monitor,
        x,
        y,
        scale: cfg.scale,
    };
    *EDIT_SESSION.lock().unwrap() = Some(session);
    let cover = (|| -> tauri::Result<()> {
        win.set_position(LogicalPosition::new(monitor.x, monitor.y))?;
        win.set_size(LogicalSize::new(monitor.width, monitor.height))?;
        win.set_ignore_cursor_events(false)
    })();
    if let Err(e) = cover {
        *EDIT_SESSION.lock().unwrap() = None;
        reconcile(app, cfg);
        return Err(e.to_string());
    }
    let _ = app.emit("overlay:edit", Some(session));
    Ok(())
}

/// `None` cancels. Window calls are best effort so both windows always leave
/// edit mode; a second call (Done pressed in both windows) is a no-op.
pub fn edit_end(app: &tauri::AppHandle, saved: &OverlayConfig, result: Option<EditResult>) {
    if EDIT_SESSION.lock().unwrap().take().is_none() {
        return;
    }
    if let Some(win) = app.get_webview_window(OVERLAY_LABEL) {
        if saved.enabled {
            let mut cfg = *saved;
            if let Some(r) = result {
                cfg.pos_x = Some(r.x);
                cfg.pos_y = Some(r.y);
                cfg.scale = r.scale;
            }
            let _ = win.set_ignore_cursor_events(true);
            let _ = position(&win, &cfg);
        } else {
            let _ = win.close();
        }
    }
    let _ = app.emit("overlay:edit", None::<EditSession>);
    let _ = app.emit("overlay:edit-result", result);
}
```

```tsx
const DEFAULT_OVERLAY: OverlayConfig = {
	enabled: true,
	c4_timer_s: C4_FUSE_DEFAULT,
	safety_margin_s: 0,
	pos_x: null,
	pos_y: null,
	scale: 1,

// Show the badge for a short grace period after detonation so 0.0 is visible.
const GRACE_MS = 1500;

// Reading shown in edit mode, so the user positions a realistic badge.
const SAMPLE_S = 32.5;

const endEdit = (result: OverlayEditResult | null) =>
	invoke("overlay_edit_end", { result }).catch(() => {});

export function Overlay() {
	const [cfg, setCfg] = useState<OverlayConfig>(DEFAULT_OVERLAY);
	const [session, setSession] = useState<OverlayEditSession | null>(null);
	// Debug page toggle: show the idle placeholder even without a plant.
	const [preview, setPreview] = useState(false);
	const [state, setState] = useState<GameState | null>(null);
	const [remaining, setRemaining] = useState<number | null>(null);

	// performance.now() when the bomb was planted.
	const plantedAt = useRef<number | null>(null);
	const prevBomb = useRef<string | null>(null);

	useEffect(() => {
		invoke<AppConfig>("get_app_settings").then(
			(a) => setCfg(a.overlay),
			() => {},
		);
		// A window created for the edit loads after `overlay:edit` fired.
		invoke<OverlayEditSession | null>("overlay_edit_session").then(
			setSession,
			() => {},
		);
		const unlisteners = [
			listen<OverlayConfig>("overlay:config", (e) => setCfg(e.payload)),
			listen<OverlayEditSession | null>("overlay:edit", (e) =>
				setSession(e.payload),
			),
			listen<boolean>("overlay:preview", (e) => setPreview(e.payload)),
			listen<GsiUpdate>("gsi:update", (e) => setState(e.payload.state)),
		];
		return () => {
			for (const un of unlisteners) un.then((f) => f());
		};
	}, []);

	useEffect(() => {
		const bomb = state?.bomb ?? null;
		if (bomb === "planted" && prevBomb.current !== "planted") {
			plantedAt.current = performance.now();
		} else if (bomb !== "planted" && bomb !== "defusing") {
			// "defused", "exploded", or gone.
			plantedAt.current = null;
		}
		prevBomb.current = bomb;
	}, [state?.bomb]);

	useEffect(() => {
		let raf = 0;
		const tick = () => {
			if (plantedAt.current == null) {
				setRemaining(null);
			} else {
				const elapsed = (performance.now() - plantedAt.current) / 1000;
				const left = cfg.c4_timer_s - elapsed - cfg.safety_margin_s;
				if (left <= -GRACE_MS / 1000) {
					plantedAt.current = null;
					setRemaining(null);
				} else {
					setRemaining(Math.max(0, left));
				}
			}
			raf = requestAnimationFrame(tick);
		};
		raf = requestAnimationFrame(tick);
		return () => cancelAnimationFrame(raf);
	}, [cfg.c4_timer_s, cfg.safety_margin_s]);

	if (session) return <EditSurface session={session} />;
	if (!preview && remaining == null) return null;

	const color =
		remaining == null
			? null
			: defuseColor(state?.team ?? null, state?.defusekit ?? false, remaining);

	return (
		<div className="flex h-screen w-screen select-none items-center justify-center">
			<TimerBadge
				scale={cfg.scale}
				color={color}
				value={remaining == null ? "--.-" : remaining.toFixed(1)}
				caption={
					color == null ? "bomb timer" : color === "red" ? "run" : "defuse"
				}
			/>
		</div>
	);
}

function TimerBadge({

	color,

// The window covers its monitor during an edit; client coordinates are
// logical px relative to the monitor's origin.
function EditSurface({ session }: { session: OverlayEditSession }) {
	const { monitor } = session;
	const [scale, setScale] = useState(session.scale);
	const [origin, setOrigin] = useState({ x: session.x, y: session.y });
	const box = clampToMonitor({ ...origin, ...overlaySize(scale) }, monitor);
	const bar = placeToolbar(box, monitor);
	const grab = useRef<{ dx: number; dy: number } | null>(null);

	// The main window's Done/Cancel end the edit with this window's unsaved
	// geometry.
	const latest = useRef({ box, scale });
	latest.current = { box, scale };
	useEffect(() => {
		const un = listen<"ok" | "cancel">("overlay:edit-request", (e) => {
			const { box, scale } = latest.current;
			endEdit(e.payload === "ok" ? { x: box.x, y: box.y, scale } : null);
		});
		return () => {
			un.then((f) => f());
		};
	}, []);

	const onPointerDown = (e: PointerEvent<HTMLDivElement>) => {
		if (e.button !== 0) return;
		e.currentTarget.setPointerCapture(e.pointerId);
		grab.current = {
			dx: monitor.x + e.clientX - box.x,
			dy: monitor.y + e.clientY - box.y,
		};
	};
	const onPointerMove = (e: PointerEvent<HTMLDivElement>) => {
		const g = grab.current;
		if (!g) return;
		const next = clampToMonitor(
			{
				x: monitor.x + e.clientX - g.dx,
				y: monitor.y + e.clientY - g.dy,
				...overlaySize(scale),
			},
			monitor,
		);
		setOrigin({ x: next.x, y: next.y });
	};
	const onPointerUp = () => {
		grab.current = null;
	};

	return (
		<div className="relative h-screen w-screen select-none bg-black/20">
			<div
				onPointerDown={onPointerDown}
				onPointerMove={onPointerMove}
				onPointerUp={onPointerUp}
				onPointerCancel={onPointerUp}
				className="absolute flex cursor-move items-center justify-center rounded-lg border-2 border-dashed border-white/70"
				style={{
					left: box.x - monitor.x,
					top: box.y - monitor.y,
					width: box.width,
					height: box.height,
				}}
			>
				<TimerBadge
					scale={scale}
					color="green"
					value={SAMPLE_S.toFixed(1)}
					caption={`drag to move · ${scale.toFixed(2)}x`}
				/>
			</div>
			<EditToolbar
				style={{ left: bar.x - monitor.x, top: bar.y - monitor.y }}
				onCancel={() => endEdit(null)}
				onSmaller={
					scale > SCALE_MIN ? () => setScale((s) => stepScale(s, -1)) : undefined
				}
				onLarger={
					scale < SCALE_MAX ? () => setScale((s) => stepScale(s, 1)) : undefined
				}
				onSave={() => endEdit({ x: box.x, y: box.y, scale })}
			/>
		</div>
	);
}
```

```tsx
// visit to the Debug page.
export function SettingsProvider({ children }: { children: ReactNode }) {
	const [store] = useState(() =>
		createSettingsStore<Settings>({
			save: async (s) => {
				await invoke("save_config", { config: s.config, app: s.app });
			},
			notify: showNotice,
		}),
	);
	const { current, history } = useSyncExternalStore(
		store.subscribe,
		store.getSnapshot,
	);
	const [loadError, setLoadError] = useState<string | null>(null);
	const [overlayEditing, setOverlayEditing] = useState(false);

	const load = useCallback(async () => {
		setLoadError(null);
		try {
			const [snap, app] = await Promise.all([
				invoke<EngineSnapshot>("engine_status"),
				invoke<AppConfig>("get_app_settings"),
			]);
			if (!snap.config) throw new Error("the engine has no config yet");
			store.replace({ config: snap.config, app });
		} catch (e) {
			setLoadError(String(e));
		}
	}, [store]);

	useEffect(() => {
		void load();
		const unlisteners = [
			listen<Settings>("settings:changed", (e) => store.replace(e.payload)),
			listen<OverlayEditSession | null>("overlay:edit", (e) =>
				setOverlayEditing(e.payload != null),
			),
			listen<OverlayEditResult | null>("overlay:edit-result", (e) => {
				const r = e.payload;
				if (!r) return;
				void store.commit(
					"Overlay position and size",
					withOverlay((o) => ({ ...o, pos_x: r.x, pos_y: r.y, scale: r.scale })),
				);
			}),
		];
		invoke<OverlayEditSession | null>("overlay_edit_session").then(
			(s) => setOverlayEditing(s != null),
			() => {},
		);
		return () => {
			for (const un of unlisteners) un.then((fn) => fn());
		};
	}, [load, store]);

	const value = useMemo<SettingsContextValue>(
		() => ({
			settings: current,
			loadError,
			reload: () => void load(),
			commit: (label, update) => void store.commit(label, update),
			resetDefaults: () =>
				void store.run("Reset to defaults", "Reset failed", async () => {
					const { config, app } = await invoke<ConfigPayload>("reset_config");
					return { config, app };
				}),
			adoptExternal: (s) => store.replace(s),
			undo: () => void store.undo(),
			redo: () => void store.redo(),
			undoLabel:
				history.past.length > 0
					? history.past[history.past.length - 1].label
					: null,
			redoLabel: history.future.length > 0 ? history.future[0].label : null,
			overlayEditing,
		}),
		[current, history, loadError, load, store, overlayEditing],
	);

	return (
		<SettingsContext.Provider value={value}>{children}</SettingsContext.Provider>
	);
}

	if (!ctx) throw new Error("useSettings must be used inside SettingsProvider");
	return ctx;
}

	if (!ctx.settings) throw new Error("settings are not loaded yet");
	return { ...ctx, settings: ctx.settings };
}
```

```ts
import { useEffect } from "react";

// Text fields keep their native undo. The hotkey recorder swallows keys in
// the capture phase, so this bubble-phase listener never sees them.
function isTextField(target: EventTarget | null): boolean {
	return (
		target instanceof HTMLElement &&
		(target.isContentEditable ||
			target.tagName === "INPUT" ||
			target.tagName === "TEXTAREA")
	);
}

export function useUndoShortcuts(
	undo: () => void,
	redo: () => void,
	enabled: boolean,
) {
	useEffect(() => {
		if (!enabled) return;
		const onKeyDown = (e: KeyboardEvent) => {
			if (!e.ctrlKey || e.altKey || e.metaKey || isTextField(e.target)) return;
			if (e.code === "KeyZ" && !e.shiftKey) {
				e.preventDefault();
				undo();
			} else if (e.code === "KeyY" || (e.code === "KeyZ" && e.shiftKey)) {
				e.preventDefault();
				redo();
			}
		};
		window.addEventListener("keydown", onKeyDown);
		return () => window.removeEventListener("keydown", onKeyDown);
	}, [undo, redo, enabled]);
}
```

```tsx
import { createElement, useEffect, useRef } from "react";
import { toast } from "sonner";
import { warningsKey } from "@/lib/config-warnings";

const WARNINGS_TOAST_ID = "config-warnings";

export function useConfigWarningsToast(
	warnings: readonly string[],
	onReset: () => void,
) {
	const key = warningsKey(warnings);
	const shownKey = useRef("");
	// The effect must re-run only when the set changes, not on every snapshot.
	const latest = useRef({ warnings, onReset });
	latest.current = { warnings, onReset };

	useEffect(() => {
		if (key === shownKey.current) return;
		shownKey.current = key;
		if (!key) {
			toast.dismiss(WARNINGS_TOAST_ID);
			return;
		}
		toast.warning("Config file had problems", {
			id: WARNINGS_TOAST_ID,
			duration: Number.POSITIVE_INFINITY,
			closeButton: true,
			description: createElement(
				"ul",
				{ className: "mt-1 space-y-0.5 font-mono text-xs" },
				latest.current.warnings.map((w) => createElement("li", { key: w }, w)),
			),
			action: {
				label: "Reset to defaults",
				onClick: () => latest.current.onReset(),
			},
		});
	}, [key]);
}
```

```tsx
import { useEffect, useState } from "react";
import { Input } from "@/components/ui/input";
import { parsePositive } from "@/lib/parse";

// Commits on blur or Enter rather than per keystroke, so typing "1500" never
// saves 1, 15 and 150 on the way. Invalid text reverts.
export function NumberField({
	value,
	onCommit,
	step,
	unit,
	disabled = false,
	integer = false,
	ariaLabel,
}: {
	value: number;
	onCommit: (n: number) => void;
	step: number;
	unit: string;
	disabled?: boolean;
	integer?: boolean;
	ariaLabel: string;
}) {
	const [text, setText] = useState(String(value));
	useEffect(() => setText(String(value)), [value]);

	const commit = () => {
		const n = parsePositive(text, integer);
		if (n == null || n === value) {
			setText(String(value));
			return;
		}
		onCommit(n);
	};

	return (
		<div className="flex items-center gap-1">
			<Input
				type="number"
				min={integer ? 1 : 0}
				step={step}
				value={text}
				disabled={disabled}
				aria-label={ariaLabel}
				onChange={(e) => setText(e.currentTarget.value)}
				onBlur={commit}
				onKeyDown={(e) => {
					if (e.key === "Enter") e.currentTarget.blur();
					if (e.key === "Escape") setText(String(value));
				}}
				className="h-8 w-20"
			/>
			<span className="text-xs text-muted-foreground">{unit}</span>
		</div>
	);
}
```

```ts
const FRESH_MS = 10000;
const POLL_MS = 1000;

export function isFresh(gsi: GsiStatus): boolean {
	return gsi.last_payload_age_ms != null && gsi.last_payload_age_ms < FRESH_MS;
}

export function useGsiStatus() {
	const [gsi, setGsi] = useState<GsiStatus | null>(null);
	const refresh = useCallback(() => {
		invoke<GsiStatus>("gsi_status").then(setGsi, () => {});
	}, []);
	useEffect(() => {
		refresh();
		const timer = setInterval(refresh, POLL_MS);
		return () => clearInterval(timer);
	}, [refresh]);
	return { gsi, refresh };
}
```

```tsx
import { invoke } from "@tauri-apps/api/core";
import { useEffect, useRef, useState } from "react";
import { Button } from "@/components/ui/button";
import { toastError } from "@/lib/toast";

const MODIFIER_CODE = /^(Control|Shift|Alt|Meta|OS)(Left|Right)?$/;

// Uses the physical `code` so the shortcut does not depend on keyboard layout.
// Returns null for a bare modifier press (keep waiting for the real key).
function shortcutFromEvent(e: KeyboardEvent): string | null {
	if (MODIFIER_CODE.test(e.code)) return null;
	const key = e.code.replace(/^Key/, "").replace(/^Digit/, "");
	const mods: string[] = [];
	if (e.ctrlKey) mods.push("Ctrl");
	if (e.altKey) mods.push("Alt");
	if (e.shiftKey) mods.push("Shift");
	if (e.metaKey) mods.push("Super");
	return [...mods, key].join("+");
}

// The live hotkey is unregistered while recording (otherwise pressing it would
// toggle pause instead of reaching the webview) and re-registered from the
// saved config afterwards.
export function HotkeyRecorder({
	value,
	disabled,
	onChange,
}: {
	value: string;
	disabled: boolean;
	onChange: (key: string) => void;
}) {
	const [recording, setRecording] = useState(false);

	useEffect(() => {
		if (!recording) return;
		const stop = () => {
			setRecording(false);
			invoke("hotkey_capture_end").catch(() => {});
		};
		const onKeyDown = (e: KeyboardEvent) => {
			e.preventDefault();
			e.stopPropagation();
			if (e.code === "Escape") return stop();
			const shortcut = shortcutFromEvent(e);
			if (shortcut == null) return;
			onChange(shortcut);
			stop();
		};
		window.addEventListener("keydown", onKeyDown, true);
		window.addEventListener("blur", stop);
		return () => {
			window.removeEventListener("keydown", onKeyDown, true);
			window.removeEventListener("blur", stop);
		};
	}, [recording, onChange]);

	// Leaving the page mid-recording must not leave the hotkey unregistered.
	const recordingRef = useRef(false);
	recordingRef.current = recording;
	useEffect(
		() => () => {
			if (recordingRef.current) invoke("hotkey_capture_end").catch(() => {});
		},
		[],
	);

	const start = async () => {
		try {
			await invoke("hotkey_capture_start");
			setRecording(true);
		} catch (e) {
			toastError("Couldn't record hotkey", e);
		}
	};

	return (
		<Button
			variant={recording ? "secondary" : "outline"}
			disabled={disabled}
			onClick={recording ? undefined : start}
			// Fixed width: the label changes while recording.
			className="w-48 font-mono"
		>
			{recording ? "Press a key… (Esc)" : value}
		</Button>
	);
}
```
